function searchUrl(ex){
  const clean = ex.n.replace(/（.*?）/g,'').replace(/→/g,' ').trim();
  const q = VQ[ex.id] || (['punch','def','foot'].includes(ex.c) ? 'ボクシング '+clean : clean+' やり方');
  return 'https://www.youtube.com/results?search_query=' + encodeURIComponent(q);
}

function setVideo(ex, after){
  const cur = VIDS[ex.id] || '';
  const msg = '【'+ex.n+'】のお気に入り動画\n\n'
            + 'YouTubeで良かった動画のURLを貼り付けてください。\n'
            + '例）https://youtu.be/xxxxxxxx\n\n'
            + (cur ? '※ 空っぽにして「OK」を押すと登録を解除します。' : '');
  const v = prompt(msg, cur);
  if(v === null) return;                       // キャンセル
  const t = v.trim();
  if(!t){ delete VIDS[ex.id]; saveVids(); after && after(); return; }
  if(!/^https?:\/\//i.test(t)){ alert('「https://」で始まるURLを貼り付けてください。'); return; }
  VIDS[ex.id] = t; saveVids(); after && after();
}

function renderStats(){
  const l = getLog(), t = today();
  const td = l.filter(x=>x.date===t);
  document.getElementById('sToday').textContent = td.reduce((a,b)=>a+b.rounds,0);
  document.getElementById('sMin').textContent = Math.round(td.reduce((a,b)=>a+b.sec,0)/60);
  // 連続日数
  const days = [...new Set(l.map(x=>x.date))].sort().reverse();
  let st=0, cur=new Date();
  for(const d of days){
    if(d === cur.toLocaleDateString('sv-SE')){ st++; cur.setDate(cur.getDate()-1); }
    else break;
  }
  document.getElementById('sStreak').textContent = st;
}
