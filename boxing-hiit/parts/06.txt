let actx=null;
function audio(){ if(!actx) actx = new (window.AudioContext||window.webkitAudioContext)(); if(actx.state==='suspended') actx.resume(); return actx; }
function beep(freq=880, dur=.12, vol=.28){
  if(!S.beep) return;
  try{
    const a=audio(), o=a.createOscillator(), g=a.createGain();
    o.type='sine'; o.frequency.value=freq;
    g.gain.setValueAtTime(vol, a.currentTime);
    g.gain.exponentialRampToValueAtTime(.001, a.currentTime+dur);
    o.connect(g); g.connect(a.destination); o.start(); o.stop(a.currentTime+dur);
  }catch(e){}
}
const beepGo   = ()=>{ beep(1100,.13); setTimeout(()=>beep(1400,.18),140); };
const beepStop = ()=>{ beep(440,.3,.24); };
const beepTick = ()=>{ beep(760,.07,.2); };

function say(txt, rate){
  if(!S.voice || !('speechSynthesis' in window)) return;
  try{
    speechSynthesis.cancel();
    const u=new SpeechSynthesisUtterance(txt);
    u.lang='ja-JP'; u.rate=rate||1.2; u.pitch=1;
    speechSynthesis.speak(u);
  }catch(e){}
}
