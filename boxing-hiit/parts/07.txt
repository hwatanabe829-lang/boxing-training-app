function readable(t){ return t.replace(/→/g,'、').replace(/[／・]/g,'、').replace(/～/g,'から'); }
function sayEx(ex, head){
  if(!S.tips){ say((head||'')+readable(ex.n)); return; }
  say((head||'') + readable(ex.n) + '。' + readable(ex.d), 1.1);
}

let wl=null;

function unlockScreen(){ try{ wl && wl.release(); wl=null; }catch(e){} }
document.addEventListener('visibilitychange', ()=>{ if(document.visibilityState==='visible' && run.classList.contains('on')) lockScreen(); });

function shuffle(a){ a=[...a]; for(let i=a.length-1;i>0;i--){const j=(Math.random()*(i+1))|0;[a[i],a[j]]=[a[j],a[i]];} return a; }

function buildSteps(p){
  const per = p.r || 8;
  let base;
  if(p.l) base = p.l;
  else {                                   // ランダム：カテゴリが偏らないよう混ぜる
    const pick = c => shuffle(EX.filter(e=>e.c===c)).map(e=>e.id);
    const pools = [pick('punch'),pick('leg'),pick('core'),pick('foot'),pick('def')];
    base=[]; let i=0;
    while(base.length < per){ const q=pools[i%pools.length]; if(q.length) base.push(q.pop()); i++; }
  }
  const steps=[];
  steps.push({type:'prep', dur:10});
  for(let s=0; s<S.sets; s++){
    for(let r=0; r<per; r++){
      steps.push({type:'work', dur:S.work, ex:EXMAP[base[(s*per+r)%base.length]]});
      const last = (s===S.sets-1 && r===per-1);
      if(!last){
        if(r===per-1) steps.push({type:'setrest', dur:S.setrest});
        else steps.push({type:'rest', dur:S.rest});
      }
    }
  }
  return steps;
}

function start(p){
  audio(); // ユーザー操作のタイミングで音を有効化
  ses = {p, steps:buildSteps(p), i:0, left:0, paused:false, spent:0, lastSay:-1};
  home.classList.remove('on'); fin.classList.remove('on');
  run.classList.add('on');
  lockScreen();
  enter(0);
  clearInterval(ses.timer);
  ses.timer = setInterval(loop, 100);
}
